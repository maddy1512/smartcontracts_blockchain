import torch
import torch.nn as nn
from torch import Tensor
from typing import Optional, Tuple
from torch_geometric.nn import MessagePassing
from torch_geometric.nn.dense.linear import Linear
from torch_geometric.utils import scatter

class BEANConv(MessagePassing):
    def __init__(
        self,
        in_channels: Tuple[int, int, Optional[int]],
        out_channels: Tuple[int, int, Optional[int]],
        node_self_loop: bool = True,
        normalize: bool = True,
        bias: bool = True,
    ):
        super().__init__(aggr=None)  # We'll control aggregation explicitly

        self.in_channels = in_channels
        self.out_channels = out_channels
        self.node_self_loop = node_self_loop
        self.normalize = normalize

        self.input_has_edge_channel = len(in_channels) == 3
        self.output_has_edge_channel = len(out_channels) == 3

        # --- Calculate input dims ---
        if self.input_has_edge_channel:
            if self.node_self_loop:
                self.in_channels_u = in_channels[0] + 2 * in_channels[1] + 2 * in_channels[2]
                self.in_channels_v = 2 * in_channels[0] + in_channels[1] + 2 * in_channels[2]
            else:
                self.in_channels_u = 2 * in_channels[1] + 2 * in_channels[2]
                self.in_channels_v = 2 * in_channels[0] + 2 * in_channels[2]
            self.in_channels_e = in_channels[0] + in_channels[1] + in_channels[2]
        else:
            if self.node_self_loop:
                self.in_channels_u = in_channels[0] + 2 * in_channels[1]
                self.in_channels_v = 2 * in_channels[0] + in_channels[1]
            else:
                self.in_channels_u = 2 * in_channels[1]
                self.in_channels_v = 2 * in_channels[0]
            self.in_channels_e = in_channels[0] + in_channels[1]

        # Linear transforms
        self.lin_u = Linear(self.in_channels_u, out_channels[0], bias=bias)
        self.lin_v = Linear(self.in_channels_v, out_channels[1], bias=bias)
        if self.output_has_edge_channel:
            self.lin_e = Linear(self.in_channels_e, out_channels[2], bias=bias)

        # BatchNorm
        if normalize:
            self.bn_u = nn.BatchNorm1d(out_channels[0])
            self.bn_v = nn.BatchNorm1d(out_channels[1])
            if self.output_has_edge_channel:
                self.bn_e = nn.BatchNorm1d(out_channels[2])

        self.reset_parameters()

    def reset_parameters(self):
        self.lin_u.reset_parameters()
        self.lin_v.reset_parameters()
        if self.output_has_edge_channel:
            self.lin_e.reset_parameters()
        if self.normalize:
            self.bn_u.reset_parameters()
            self.bn_v.reset_parameters()
            if self.output_has_edge_channel:
                self.bn_e.reset_parameters()

    def forward(
        self,
        x: Tuple[Tensor, Tensor],           # (xu, xv)
        edge_index: Tensor,                 # [2, E]
        edge_attr: Optional[Tensor] = None  # xe
    ) -> Tuple[Tuple[Tensor, Tensor], Optional[Tensor]]:
        (msg_2u, msg_2v), msg_2e = self.message_and_aggregate(edge_index, x, edge_attr)

        out_u = self.lin_u(msg_2u)
        out_v = self.lin_v(msg_2v)
        out_e = self.lin_e(msg_2e) if self.output_has_edge_channel and msg_2e is not None else None

        if self.normalize:
            out_u = self.bn_u(out_u)
            out_v = self.bn_v(out_v)
            if out_e is not None:
                out_e = self.bn_e(out_e)

        return (out_u, out_v), out_e

    def message_and_aggregate(
        self,
        edge_index: Tensor,
        x: Tuple[Tensor, Tensor],
        xe: Optional[Tensor] = None
    ) -> Tuple[Tuple[Tensor, Tensor], Optional[Tensor]]:

        xu, xv = x
        row, col = edge_index  # row: u indices, col: v indices

        # --- Node-to-node messages ---
        msg_v2u_mean = scatter(xv[col], row, dim=0, dim_size=xu.size(0), reduce='mean')
        msg_v2u_max  = scatter(xv[col], row, dim=0, dim_size=xu.size(0), reduce='max')
        msg_u2v_mean = scatter(xu[row], col, dim=0, dim_size=xv.size(0), reduce='mean')
        msg_u2v_max  = scatter(xu[row], col, dim=0, dim_size=xv.size(0), reduce='max')

        # --- Edge-to-node messages ---
        if xe is not None:
            msg_e2u_mean = scatter(xe, row, dim=0, dim_size=xu.size(0), reduce='mean')
            msg_e2u_max  = scatter(xe, row, dim=0, dim_size=xu.size(0), reduce='max')
            msg_e2v_mean = scatter(xe, col, dim=0, dim_size=xv.size(0), reduce='mean')
            msg_e2v_max  = scatter(xe, col, dim=0, dim_size=xv.size(0), reduce='max')

        # --- Collect messages for nodes ---
        if xe is not None:
            if self.node_self_loop:
                msg_2u = torch.cat([xu, msg_v2u_mean, msg_v2u_max, msg_e2u_mean, msg_e2u_max], dim=1)
                msg_2v = torch.cat([xv, msg_u2v_mean, msg_u2v_max, msg_e2v_mean, msg_e2v_max], dim=1)
            else:
                msg_2u = torch.cat([msg_v2u_mean, msg_v2u_max, msg_e2u_mean, msg_e2u_max], dim=1)
                msg_2v = torch.cat([msg_u2v_mean, msg_u2v_max, msg_e2v_mean, msg_e2v_max], dim=1)

            msg_2e = None
            if self.output_has_edge_channel:
                msg_2e = torch.cat([xe, xu[row], xv[col]], dim=1)
        else:
            if self.node_self_loop:
                msg_2u = torch.cat([xu, msg_v2u_mean, msg_v2u_max], dim=1)
                msg_2v = torch.cat([xv, msg_u2v_mean, msg_u2v_max], dim=1)
            else:
                msg_2u = torch.cat([msg_v2u_mean, msg_v2u_max], dim=1)
                msg_2v = torch.cat([msg_u2v_mean, msg_u2v_max], dim=1)

            msg_2e = None
            if self.output_has_edge_channel:
                msg_2e = torch.cat([xu[row], xv[col]], dim=1)

        return (msg_2u, msg_2v), msg_2e
